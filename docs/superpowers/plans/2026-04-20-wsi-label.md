# wsi-label Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `wsi-label`, a Go CLI that replaces or strips the label image inside an Aperio SVS file, without touching pyramid pixel data, while preserving the ability to add backends for other WSI formats later.

**Architecture:** "Copy prefix, re-emit tail." Parse the TIFF/BigTIFF structure, identify the label IFD by `ImageDescription`, compute a cutoff = min-offset of any byte owned by the label-or-later IFDs, copy `[0, cutoff)` verbatim, emit a new label IFD + strip data, re-emit trailing IFDs with rebased offsets, patch one `next-IFD` pointer in the copied prefix. Temp-file + rename for atomicity. Pure Go throughout.

**Tech Stack:** Go 1.22+ (1.26 installed), stdlib only for core (`encoding/binary`, `compress/lzw`, `image`, `image/png`, `image/jpeg`, `golang.org/x/image/draw` for resize, `golang.org/x/image/tiff` for decoding input TIFFs). CLI: stdlib `flag` (or `github.com/spf13/cobra` if we want richer UX — decision deferred to Task 2). Testing: stdlib `testing`, `go-cmp`, optional `openslide-python`/`tifffile` shell-outs for round-trip verification.

**Spec:** `docs/superpowers/specs/2026-04-20-wsi-label-design.md`

**Why this split:**
- `tifflow` is vendor-agnostic, reused by any future TIFF-based backend.
- `aperio` holds only Aperio-specific knowledge (label spec, detection heuristic).
- `wsi` is the seam: adding a second backend is one package + one registry entry.
- `cli` is a thin shell over `wsi`. Exit codes are the ONLY contract with shell scripts; tests assert them.
- `atomic` is tiny but reused by all output paths; isolating it means we test it once.

**Scope for this plan:** Aperio SVS only. The `wsi.Backend` interface is defined and used, but the registry has one entry (aperio) and picks it unconditionally. Adding a second backend later is a separate plan.

```go
// Package atomic provides atomic file writes via temp-file + rename.
package atomic

import (
	"fmt"
	"os"
	"path/filepath"
	"time"
)

// WriteAtomic writes to target by first writing to a sibling temp file,
// syncing (if fsync is true), then renaming over target. On any error from
// write, the temp file is removed and target is left untouched.
func WriteAtomic(target string, write func(w *os.File) error, fsync bool) error {
	dir := filepath.Dir(target)
	base := filepath.Base(target)
	tmpName := fmt.Sprintf(".%s.tmp-%d-%d", base, os.Getpid(), time.Now().UnixNano())
	tmpPath := filepath.Join(dir, tmpName)

	f, err := os.OpenFile(tmpPath, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0o644)
	if err != nil {
		return fmt.Errorf("create temp: %w", err)
	}

	cleanup := func() {
		_ = f.Close()
		_ = os.Remove(tmpPath)
	}

	if err := write(f); err != nil {
		cleanup()
		return err
	}
	if fsync {
		if err := f.Sync(); err != nil {
			cleanup()
			return fmt.Errorf("fsync temp: %w", err)
		}
	}
	if err := f.Close(); err != nil {
		_ = os.Remove(tmpPath)
		return fmt.Errorf("close temp: %w", err)
	}
	if err := os.Rename(tmpPath, target); err != nil {
		_ = os.Remove(tmpPath)
		return fmt.Errorf("rename: %w", err)
	}
	return nil
}
```

```go
package cli

import "errors"

// Sentinel errors. CLI maps these to exit codes.
var (
	ErrUsage              = errors.New("usage error")
	ErrStrictReplace      = errors.New("no existing label (--strict-replace)")
	ErrFormatUnrecognized = errors.New("unrecognized WSI format")
	ErrUnreadable         = errors.New("input unreadable")
	ErrOutputExists       = errors.New("output path exists (use --overwrite)")
	ErrUnexpectedLayout   = errors.New("input file has unexpected TIFF layout")
	ErrLabelAdded         = errors.New("label added rather than replaced")
)

// ExitCode returns the process exit code for a completed run. Nil → 0.
func ExitCode(err error) int {
	if err == nil {
		return 0
	}
	switch {
	case errors.Is(err, ErrUsage), errors.Is(err, ErrStrictReplace):
		return 2
	case errors.Is(err, ErrUnreadable), errors.Is(err, ErrFormatUnrecognized):
		return 3
	case errors.Is(err, ErrOutputExists):
		return 4
	case errors.Is(err, ErrUnexpectedLayout):
		return 5
	case errors.Is(err, ErrLabelAdded):
		return 10
	default:
		return 1
	}
}
```

```go
package cli

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// ResolveOutputPath returns the final output path for a replace/strip run.
//
//   - If out is empty, derive "<stem>_relabeled<ext>" next to input. If that
//     exists, try "_relabeled_1", "_2", ... until a free name is found (unless
//     overwrite is true, in which case the bare derived name is clobbered).
//   - If out is non-empty, use it. If it exists and overwrite is false,
//     return ErrOutputExists.
//   - In every case, if the resolved path equals the input path, return
//     ErrUsage.
func ResolveOutputPath(input, out string, overwrite bool) (string, error) {
	absInput, err := filepath.Abs(input)
	if err != nil {
		return "", fmt.Errorf("abs input: %w", err)
	}

	var resolved string
	if out == "" {
		resolved = derivePath(absInput, overwrite)
	} else {
		absOut, err := filepath.Abs(out)
		if err != nil {
			return "", fmt.Errorf("abs out: %w", err)
		}
		if _, err := os.Stat(absOut); err == nil && !overwrite {
			return "", fmt.Errorf("%w: %s", ErrOutputExists, absOut)
		}
		resolved = absOut
	}

	if resolved == absInput {
		return "", fmt.Errorf("%w: input and output paths are the same", ErrUsage)
	}
	return resolved, nil
}

func derivePath(absInput string, overwrite bool) string {
	dir := filepath.Dir(absInput)
	base := filepath.Base(absInput)
	ext := filepath.Ext(base)
	stem := strings.TrimSuffix(base, ext)

	first := filepath.Join(dir, stem+"_relabeled"+ext)
	if overwrite {
		return first
	}
	if _, err := os.Stat(first); os.IsNotExist(err) {
		return first
	}
	for i := 1; ; i++ {
		candidate := filepath.Join(dir, fmt.Sprintf("%s_relabeled_%d%s", stem, i, ext))
		if _, err := os.Stat(candidate); os.IsNotExist(err) {
			return candidate
		}
	}
}
```

```go
package tifflow

import (
	"encoding/binary"
	"fmt"
	"io"
)

// Header is the parsed TIFF / BigTIFF header.
type Header struct {
	ByteOrder      binary.ByteOrder
	BigTIFF        bool
	FirstIFDOffset uint64
}

// ParseHeader reads the 8-byte classic or 16-byte BigTIFF header from r.
func ParseHeader(r io.Reader) (*Header, error) {
	var magic [4]byte
	if _, err := io.ReadFull(r, magic[:]); err != nil {
		return nil, fmt.Errorf("read magic: %w", err)
	}
	var order binary.ByteOrder
	switch {
	case magic[0] == 'I' && magic[1] == 'I':
		order = binary.LittleEndian
	case magic[0] == 'M' && magic[1] == 'M':
		order = binary.BigEndian
	default:
		return nil, ErrBadMagic
	}
	version := order.Uint16(magic[2:4])
	h := &Header{ByteOrder: order}
	switch version {
	case 42:
		var off uint32
		if err := binary.Read(r, order, &off); err != nil {
			return nil, fmt.Errorf("read first IFD offset: %w", err)
		}
		h.FirstIFDOffset = uint64(off)
	case 43:
		h.BigTIFF = true
		var offsetSize uint16
		var reserved uint16
		if err := binary.Read(r, order, &offsetSize); err != nil {
			return nil, fmt.Errorf("read offset size: %w", err)
		}
		if offsetSize != 8 {
			return nil, fmt.Errorf("%w: BigTIFF offset size = %d, want 8", ErrBadMagic, offsetSize)
		}
		if err := binary.Read(r, order, &reserved); err != nil {
			return nil, fmt.Errorf("read reserved: %w", err)
		}
		if reserved != 0 {
			return nil, fmt.Errorf("%w: BigTIFF reserved = %d, want 0", ErrBadMagic, reserved)
		}
		if err := binary.Read(r, order, &h.FirstIFDOffset); err != nil {
			return nil, fmt.Errorf("read first IFD offset: %w", err)
		}
	default:
		return nil, fmt.Errorf("%w: version = %d", ErrBadMagic, version)
	}
	return h, nil
}

// EmitHeader writes the header in the same variant as h describes.
func EmitHeader(w io.Writer, h *Header) error {
	var magic [4]byte
	if h.ByteOrder == binary.LittleEndian {
		magic[0], magic[1] = 'I', 'I'
	} else {
		magic[0], magic[1] = 'M', 'M'
	}
	if h.BigTIFF {
		h.ByteOrder.PutUint16(magic[2:4], 43)
	} else {
		h.ByteOrder.PutUint16(magic[2:4], 42)
	}
	if _, err := w.Write(magic[:]); err != nil {
		return err
	}
	if h.BigTIFF {
		if err := binary.Write(w, h.ByteOrder, uint16(8)); err != nil {
			return err
		}
		if err := binary.Write(w, h.ByteOrder, uint16(0)); err != nil {
			return err
		}
		return binary.Write(w, h.ByteOrder, h.FirstIFDOffset)
	}
	return binary.Write(w, h.ByteOrder, uint32(h.FirstIFDOffset))
}

// HeaderSize returns the number of bytes the header occupies in the file.
func HeaderSize(bigTIFF bool) int64 {
	if bigTIFF {
		return 16
	}
	return 8
}
```

```go
import (
```

```go
// IFDEntry is a single tag entry in an IFD.
type IFDEntry struct {
	Tag   uint16
	Type  TagType
	Count uint64

	// RawValueField holds the 4-byte (classic) or 8-byte (BigTIFF) value field
	// exactly as it appeared in the IFD record. For inline values, the bytes
	// encode the value directly. For out-of-line values, it encodes the offset.
	RawValueField []byte

	// Inline is true when the value fits in RawValueField; false when
	// RawValueField encodes an offset to out-of-line data.
	Inline bool

	// DataOffset is the out-of-line offset (when Inline is false).
	DataOffset uint64

	// DataSize is the byte length of the value data (Count * Type.Size()),
	// computed for convenience.
	DataSize uint64

	// InlineValue is the raw out-of-line bytes (when Inline is false,
	// populated by ReadIFD), or a copy of RawValueField trimmed to DataSize
	// (when Inline is true).
	InlineValue []byte
}

// IFD is a parsed TIFF image file directory.
type IFD struct {
	// Offset where this IFD record begins in the file.
	Offset uint64
	// Length of the IFD record in bytes (not including out-of-line data).
	RecordLength uint64
	// Position of the next-IFD pointer inside the file.
	NextPointerOffset uint64
	// Value of the next-IFD pointer (0 = end of chain).
	NextOffset uint64

	Entries []IFDEntry
}

// ReadIFD reads one IFD record at the given offset. It fetches out-of-line
// data for every entry whose DataSize is small enough to be useful for
// parsing (<=4096 bytes), primarily so callers can read ImageDescription.
// Larger out-of-line blobs (big StripOffsets arrays for multi-GB files) are
// NOT read — callers that need them use ReadOutOfLine.
func ReadIFD(r io.ReaderAt, h *Header, offset uint64) (*IFD, error) {
	ifd := &IFD{Offset: offset}

	// Entry count: 2 bytes (classic) or 8 bytes (BigTIFF).
	var countBuf [8]byte
	countSize := 2
	if h.BigTIFF {
		countSize = 8
	}
	if _, err := r.ReadAt(countBuf[:countSize], int64(offset)); err != nil {
		return nil, fmt.Errorf("read entry count: %w", err)
	}
	var count uint64
	if h.BigTIFF {
		count = h.ByteOrder.Uint64(countBuf[:8])
	} else {
		count = uint64(h.ByteOrder.Uint16(countBuf[:2]))
	}

	// Each entry: 12 bytes (classic) or 20 bytes (BigTIFF).
	entrySize := 12
	if h.BigTIFF {
		entrySize = 20
	}
	entriesSize := int(count) * entrySize
	entriesBuf := make([]byte, entriesSize)
	if _, err := r.ReadAt(entriesBuf, int64(offset)+int64(countSize)); err != nil {
		return nil, fmt.Errorf("read entries: %w", err)
	}

	valueFieldSize := 4
	if h.BigTIFF {
		valueFieldSize = 8
	}

	entries := make([]IFDEntry, count)
	for i := uint64(0); i < count; i++ {
		base := int(i) * entrySize
		e := &entries[i]
		e.Tag = h.ByteOrder.Uint16(entriesBuf[base : base+2])
		e.Type = TagType(h.ByteOrder.Uint16(entriesBuf[base+2 : base+4]))
		if h.BigTIFF {
			e.Count = h.ByteOrder.Uint64(entriesBuf[base+4 : base+12])
		} else {
			e.Count = uint64(h.ByteOrder.Uint32(entriesBuf[base+4 : base+8]))
		}
		valStart := base + 4 + (entrySize - 4 - valueFieldSize) + valueFieldSize
		// Above: for classic, base+8..base+12. For BigTIFF, base+12..base+20.
		var rawVal []byte
		if h.BigTIFF {
			rawVal = make([]byte, 8)
			copy(rawVal, entriesBuf[base+12:base+20])
		} else {
			rawVal = make([]byte, 4)
			copy(rawVal, entriesBuf[base+8:base+12])
		}
		e.RawValueField = rawVal
		sz := e.Type.Size()
		if sz == 0 {
			return nil, fmt.Errorf("%w: type=%d", ErrUnknownType, e.Type)
		}
		e.DataSize = e.Count * uint64(sz)
		if e.DataSize <= uint64(valueFieldSize) {
			e.Inline = true
			e.InlineValue = rawVal[:e.DataSize]
		} else {
			e.Inline = false
			if h.BigTIFF {
				e.DataOffset = h.ByteOrder.Uint64(rawVal)
			} else {
				e.DataOffset = uint64(h.ByteOrder.Uint32(rawVal))
			}
			// Fetch out-of-line data if small.
			if e.DataSize <= 4096 {
				buf := make([]byte, e.DataSize)
				if _, err := r.ReadAt(buf, int64(e.DataOffset)); err != nil {
					return nil, fmt.Errorf("read out-of-line tag %d: %w", e.Tag, err)
				}
				e.InlineValue = buf
			}
		}
		_ = valStart
	}
	ifd.Entries = entries
	ifd.RecordLength = uint64(countSize + entriesSize + valueFieldSize)
	// (valueFieldSize bytes = next-IFD pointer)
	nextPointerOffset := int64(offset) + int64(countSize) + int64(entriesSize)
	ifd.NextPointerOffset = uint64(nextPointerOffset)

	nextBuf := make([]byte, valueFieldSize)
	if _, err := r.ReadAt(nextBuf, nextPointerOffset); err != nil {
		return nil, fmt.Errorf("read next IFD: %w", err)
	}
	if h.BigTIFF {
		ifd.NextOffset = h.ByteOrder.Uint64(nextBuf)
	} else {
		ifd.NextOffset = uint64(h.ByteOrder.Uint32(nextBuf))
	}

	return ifd, nil
}

// UintValue returns an unsigned integer value for tag, reading only the
// first element if multi-valued. Returns (0, false) if not present or not
// an integer type.
func (ifd *IFD) UintValue(tag uint16) (uint64, bool) {
	for _, e := range ifd.Entries {
		if e.Tag != tag {
			continue
		}
		if len(e.InlineValue) == 0 {
			return 0, false
		}
		order := binary.LittleEndian // placeholder; we need the real byte order
		_ = order
		// Values are stored in the file's byte order, but InlineValue
		// captured raw bytes. We don't have the order here; callers who
		// need proper decoding should use ReadUintAt. For simple uses
		// (ImageWidth etc.) we use the helper below.
		switch e.Type {
		case TypeByte, TypeSByte, TypeUndefined:
			return uint64(e.InlineValue[0]), true
		case TypeShort, TypeSShort:
			return uint64(binary.LittleEndian.Uint16(e.InlineValue[:2])), true
		case TypeLong, TypeSLong:
			return uint64(binary.LittleEndian.Uint32(e.InlineValue[:4])), true
		case TypeLong8, TypeSLong8, TypeIFD8:
			return binary.LittleEndian.Uint64(e.InlineValue[:8]), true
		}
		return 0, false
	}
	return 0, false
}

// StringValue returns the ASCII value of a tag, stripped of trailing NULs.
func (ifd *IFD) StringValue(tag uint16) (string, bool) {
	for _, e := range ifd.Entries {
		if e.Tag != tag {
			continue
		}
		if e.Type != TypeASCII {
			return "", false
		}
		return string(bytes.TrimRight(e.InlineValue, "\x00")), true
	}
	return "", false
}
```

```go
package tifflow

import (
	"fmt"
	"io"
)

// File is a parsed TIFF / BigTIFF file structure (metadata only, no pixel data).
type File struct {
	Header *Header
	IFDs   []*IFD
	// ChainOrder maps IFD offset → chain index (0-based). Useful for
	// computing "IFDs after label in chain" without re-scanning.
	ChainOrder map[uint64]int
}

// Parse walks the IFD chain from the header. It does not dereference
// SubIFDs (tag 330); if any IFD has SubIFDs, it returns ErrSubIFDs.
func Parse(r io.ReaderAt) (*File, error) {
	rs, ok := r.(io.ReadSeeker)
	if ok {
		if _, err := rs.Seek(0, io.SeekStart); err != nil {
			return nil, err
		}
	}

	// For the header read we need a Reader; use io.NewSectionReader.
	sz := int64(1 << 62)
	if sizer, ok := r.(interface{ Size() int64 }); ok {
		sz = sizer.Size()
	} else if stat, ok := r.(interface{ Stat() (interface{}, error) }); ok {
		_ = stat
	}
	sr := io.NewSectionReader(r, 0, sz)
	h, err := ParseHeader(sr)
	if err != nil {
		return nil, err
	}
	f := &File{Header: h, ChainOrder: map[uint64]int{}}

	seen := map[uint64]bool{}
	offset := h.FirstIFDOffset
	for offset != 0 {
		if seen[offset] {
			return nil, fmt.Errorf("%w: IFD chain loop at offset %d", ErrUnexpectedLayout, offset)
		}
		seen[offset] = true
		ifd, err := ReadIFD(r, h, offset)
		if err != nil {
			return nil, fmt.Errorf("read IFD at %d: %w", offset, err)
		}
		// Reject SubIFDs.
		for _, e := range ifd.Entries {
			if e.Tag == TagSubIFDs {
				return nil, ErrSubIFDs
			}
		}
		f.ChainOrder[offset] = len(f.IFDs)
		f.IFDs = append(f.IFDs, ifd)
		offset = ifd.NextOffset
	}
	return f, nil
}
```

```go
	Start uint64
	End   uint64
	Owner int
	What  string
}

// RangeMap holds non-overlapping byte ranges, kept sorted by Start.
type RangeMap struct {
	ranges []Range
}

// Add inserts r, returning ErrOverlap if it overlaps any existing range.
func (m *RangeMap) Add(r Range) error {
	if r.End <= r.Start {
		return fmt.Errorf("invalid range [%d, %d)", r.Start, r.End)
	}
	idx := sort.Search(len(m.ranges), func(i int) bool {
		return m.ranges[i].Start >= r.Start
	})
	if idx > 0 && m.ranges[idx-1].End > r.Start {
		return fmt.Errorf("%w: new %s[%d,%d) overlaps existing %s[%d,%d)",
			ErrOverlap, r.What, r.Start, r.End,
			m.ranges[idx-1].What, m.ranges[idx-1].Start, m.ranges[idx-1].End)
	}
	if idx < len(m.ranges) && m.ranges[idx].Start < r.End {
		return fmt.Errorf("%w: new %s[%d,%d) overlaps existing %s[%d,%d)",
			ErrOverlap, r.What, r.Start, r.End,
			m.ranges[idx].What, m.ranges[idx].Start, m.ranges[idx].End)
	}
	m.ranges = append(m.ranges, Range{})
	copy(m.ranges[idx+1:], m.ranges[idx:])
	m.ranges[idx] = r
	return nil
}

// All returns all ranges in sorted order.
func (m *RangeMap) All() []Range { return m.ranges }

// MinOffsetOfOwner returns the lowest start offset among ranges owned by
// owner. Returns ^uint64(0) if no ranges for owner.
func (m *RangeMap) MinOffsetOfOwner(owner int) uint64 {
	min := ^uint64(0)
	for _, r := range m.ranges {
		if r.Owner == owner && r.Start < min {
			min = r.Start
		}
	}
	return min
}

// MinOffsetOfOwnersAtOrAfter returns the lowest start offset among ranges
// owned by any owner >= minOwner.
func (m *RangeMap) MinOffsetOfOwnersAtOrAfter(minOwner int) uint64 {
	min := ^uint64(0)
	for _, r := range m.ranges {
		if r.Owner >= minOwner && r.Start < min {
			min = r.Start
		}
	}
	return min
}

// AnyRangeOfOwnerAtOrAfter returns the first range (by start offset) owned
// by owner whose start is >= floor.
func (m *RangeMap) AnyRangeOfOwnerAtOrAfter(owner int, floor uint64) (Range, bool) {
	for _, r := range m.ranges {
		if r.Owner == owner && r.Start >= floor {
			return r, true
		}
	}
	return Range{}, false
}
```

```go
type File struct {
	Header     *Header
	IFDs       []*IFD
	ChainOrder map[uint64]int
	Ranges     *RangeMap
}

func Parse(r io.ReaderAt) (*File, error) {
	sz := int64(1 << 62)
	if sizer, ok := r.(interface{ Size() int64 }); ok {
		sz = sizer.Size()
	}
	sr := io.NewSectionReader(r, 0, sz)
	h, err := ParseHeader(sr)
	if err != nil {
		return nil, err
	}
	f := &File{Header: h, ChainOrder: map[uint64]int{}, Ranges: &RangeMap{}}
	// Header owns its own range (no IFD owns it, so use owner = -1).
	if err := f.Ranges.Add(Range{Start: 0, End: uint64(HeaderSize(h.BigTIFF)), Owner: -1, What: "header"}); err != nil {
		return nil, err
	}

	seen := map[uint64]bool{}
	offset := h.FirstIFDOffset
	for offset != 0 {
		if seen[offset] {
			return nil, fmt.Errorf("%w: IFD chain loop at offset %d", ErrUnexpectedLayout, offset)
		}
		seen[offset] = true
		ifd, err := ReadIFD(r, h, offset)
		if err != nil {
			return nil, fmt.Errorf("read IFD at %d: %w", offset, err)
		}
		for _, e := range ifd.Entries {
			if e.Tag == TagSubIFDs {
				return nil, ErrSubIFDs
			}
		}
		owner := len(f.IFDs)
		f.ChainOrder[offset] = owner
		f.IFDs = append(f.IFDs, ifd)

		// Add IFD record range.
		if err := f.Ranges.Add(Range{
			Start: ifd.Offset,
			End:   ifd.Offset + ifd.RecordLength,
			Owner: owner,
			What:  "ifd",
		}); err != nil {
			return nil, fmt.Errorf("ifd %d range: %w", owner, err)
		}

		// Add out-of-line ranges for tags.
		for _, e := range ifd.Entries {
			if e.Inline || e.DataSize == 0 {
				continue
			}
			if err := f.Ranges.Add(Range{
				Start: e.DataOffset,
				End:   e.DataOffset + e.DataSize,
				Owner: owner,
				What:  fmt.Sprintf("tag%d", e.Tag),
			}); err != nil {
				return nil, fmt.Errorf("ifd %d tag %d range: %w", owner, e.Tag, err)
			}
		}

		// Add strip / tile data ranges.
		offs, offsOK := ifd.UintArray(TagStripOffsets, h.ByteOrder)
		if !offsOK {
			offs, offsOK = ifd.UintArray(TagTileOffsets, h.ByteOrder)
		}
		counts, countsOK := ifd.UintArray(TagStripByteCounts, h.ByteOrder)
		if !countsOK {
			counts, countsOK = ifd.UintArray(TagTileByteCounts, h.ByteOrder)
		}
		if offsOK && countsOK {
			if len(offs) != len(counts) {
				return nil, fmt.Errorf("%w: ifd %d has %d strip offsets but %d byte counts",
					ErrUnexpectedLayout, owner, len(offs), len(counts))
			}
			for i, off := range offs {
				if counts[i] == 0 {
					continue
				}
				if err := f.Ranges.Add(Range{
					Start: off,
					End:   off + counts[i],
					Owner: owner,
					What:  fmt.Sprintf("strip[%d]", i),
				}); err != nil {
					return nil, fmt.Errorf("ifd %d strip %d: %w", owner, i, err)
				}
			}
		}

		offset = ifd.NextOffset
	}
	return f, nil
}
```

```go
	if len(args) == 0 {

		return fmt.Errorf("%w: unknown subcommand %q", ErrUsage, sub)
	}
```

```go
// for Aperio SVS. It lives in cli for display purposes; wsi-label's actual
// label location is computed in the aperio backend.
func identifyRole(idx int, ifd *tifflow.IFD, order interface{}) string {
	desc, _ := ifd.StringValue(tifflow.TagImageDescription)
	lines := strings.SplitN(desc, "\n", 3)
	second := ""
	if len(lines) >= 2 {
		second = lines[1]

	case idx == 0:
		return "baseline"
	case idx == 1 && !hasTiles(ifd):
		return "thumbnail"
	case strings.HasPrefix(second, "label"):
		return "label"
	case strings.HasPrefix(second, "macro"):
		return "macro"
	case hasTiles(ifd):
		return "pyramid"
	default:
		return "unknown"
	}
}

		if e.Tag == tifflow.TagTileOffsets {
			return true
		}
	}
	return false
```

```go
package aperio

import (
	"errors"
	"os"
	"strings"

	"github.com/tcornish/wsi-label/internal/tifflow"
)

// Detect returns true if the file at path is an Aperio SVS.
// Heuristic: a valid TIFF / BigTIFF whose IFD 0 ImageDescription begins
// with "Aperio".
func Detect(path string) (bool, error) {
	f, err := os.Open(path)
	if err != nil {
		return false, err
	}
	defer f.Close()

	file, err := tifflow.Parse(f)
	if err != nil {
		if errors.Is(err, tifflow.ErrBadMagic) {
			return false, nil
		}
		return false, err
	}
	if len(file.IFDs) == 0 {
		return false, nil
	}
	desc, ok := file.IFDs[0].StringValue(tifflow.TagImageDescription)
	if !ok {
		return false, nil
	}
	return strings.HasPrefix(desc, "Aperio"), nil
}

// FindLabelIFD returns the chain-order index of the label IFD, or -1 if
// the file has no label. The rule (matching OpenSlide): non-tiled IFD
// that is not IFD 1 (thumbnail) and whose ImageDescription second line
// begins with "label".
func FindLabelIFD(file *tifflow.File) int {
	for i, ifd := range file.IFDs {
		if i == 1 {
			continue
		}
		if hasTileTags(ifd) {
			continue
		}
		desc, ok := ifd.StringValue(tifflow.TagImageDescription)
		if !ok {
			continue
		}
		lines := strings.SplitN(desc, "\n", 3)
		if len(lines) < 2 {
			continue
		}
		if strings.HasPrefix(lines[1], "label") {
			return i
		}
	}
	return -1
}

// FindMacroIFD returns the chain-order index of the macro IFD, or -1.
func FindMacroIFD(file *tifflow.File) int {
	for i, ifd := range file.IFDs {
		if i == 1 {
			continue
		}
		if hasTileTags(ifd) {
			continue
		}
		desc, ok := ifd.StringValue(tifflow.TagImageDescription)
		if !ok {
			continue
		}
		lines := strings.SplitN(desc, "\n", 3)
		if len(lines) < 2 {
			continue
		}
		if strings.HasPrefix(lines[1], "macro") {
			return i
		}
	}
	return -1
}
```

```go
package aperio

import (
	"image"
	"image/color"
	"image/draw"

	xdraw "golang.org/x/image/draw"
)

// FitTo returns a width×height RGBA image containing src aspect-preservingly
// scaled to fit, with letterbox area filled with bg.
func FitTo(src image.Image, width, height int, bg color.RGBA) *image.RGBA {
	dst := image.NewRGBA(image.Rect(0, 0, width, height))
	draw.Draw(dst, dst.Bounds(), &image.Uniform{C: bg}, image.Point{}, draw.Src)

	sb := src.Bounds()
	sw, sh := sb.Dx(), sb.Dy()
	if sw == width && sh == height {
		draw.Draw(dst, dst.Bounds(), src, sb.Min, draw.Src)
		return dst
	}

	scale := float64(width) / float64(sw)
	if s := float64(height) / float64(sh); s < scale {
		scale = s
	}
	dw := int(float64(sw) * scale)
	dh := int(float64(sh) * scale)
	ox := (width - dw) / 2
	oy := (height - dh) / 2
	target := image.Rect(ox, oy, ox+dw, oy+dh)
```

```go
import (
```

```go
// Target dimensions for Aperio SVS labels.
const (
	AperioLabelWidth  = 1200
	AperioLabelHeight = 848
	AperioRowsPerStrip = 2
)

// AperioLabelDescription is the ImageDescription written for a replacement label.
const AperioLabelDescription = "Aperio Image Library v12.4.7\nlabel 1200x848"

// AperioLabelBG is the default letterbox fill color.
var AperioLabelBG = color.RGBA{245, 245, 230, 255}

// rgbStripBytes returns tightly packed 8-bit RGB bytes for the given image,
// stride = width*3, no alpha.
func rgbStripBytes(img image.Image) []byte {
	b := img.Bounds()
	w, h := b.Dx(), b.Dy()
	out := make([]byte, w*h*3)
	for y := 0; y < h; y++ {
		for x := 0; x < w; x++ {
			r, g, bl, _ := img.At(b.Min.X+x, b.Min.Y+y).RGBA()
			out[(y*w+x)*3+0] = uint8(r >> 8)
			out[(y*w+x)*3+1] = uint8(g >> 8)
			out[(y*w+x)*3+2] = uint8(bl >> 8)
		}
	}
	return out
}

// applyPredictor2 performs TIFF horizontal differencing on RGB bytes.
// First pixel of each row is unchanged; subsequent pixels store the
// difference from the previous pixel, per channel, mod 256.
func applyPredictor2(rgb []byte, w, h int) []byte {
	out := make([]byte, len(rgb))
	stride := w * 3
	for y := 0; y < h; y++ {
		row := rgb[y*stride : (y+1)*stride]
		orow := out[y*stride : (y+1)*stride]
		copy(orow[:3], row[:3])
		for x := 1; x < w; x++ {
			for c := 0; c < 3; c++ {
				orow[x*3+c] = row[x*3+c] - row[(x-1)*3+c]
			}
		}
	}
	return out
}

// encodeLZW compresses data with TIFF-flavored LZW (MSB order, 8-bit literal).
func encodeLZW(data []byte) []byte {
	var buf bytes.Buffer
	w := lzw.NewWriter(&buf, lzw.MSB, 8)
	_, _ = w.Write(data)
	_ = w.Close()
	return buf.Bytes()
}

// EncodeLabelStrips splits an Aperio-dimensioned RGB image into
// AperioRowsPerStrip-row strips, applies predictor 2, and LZW-compresses
// each strip. Returns one compressed blob per strip.
func EncodeLabelStrips(img image.Image) [][]byte {
	b := img.Bounds()
	w, h := b.Dx(), b.Dy()
	raw := rgbStripBytes(img)
	stride := w * 3

	numStrips := (h + AperioRowsPerStrip - 1) / AperioRowsPerStrip
	strips := make([][]byte, numStrips)
	for s := 0; s < numStrips; s++ {
		y0 := s * AperioRowsPerStrip
		y1 := y0 + AperioRowsPerStrip
		if y1 > h {
			y1 = h
		}
		stripRaw := raw[y0*stride : y1*stride]
		stripPre := applyPredictor2(stripRaw, w, y1-y0)
		strips[s] = encodeLZW(stripPre)
	}
	return strips
}
```

```go
// BuildLabelIFD produces a ReplacementIFD for the given 1200×848 RGB image.
// Strip offsets are marked ResolvesToOffset so the emitter fills them in.
func BuildLabelIFD(img image.Image) *tifflow.ReplacementIFD {
	strips := EncodeLabelStrips(img)

	stripByteCounts := make([]byte, 0, len(strips)*4)
	// Byte counts written as LONG (uint32) — safe since label strips are <<4GB.
	// The emitter will re-encode these in the file's byte order. For BigTIFF,
	// we use LONG8 (uint64) via the emitter's byte-order-aware path.
	// For simplicity, we write placeholder bytes here and let the emitter
	// re-serialize per file order. We encode as little-endian LONG for now;
	// the emitter rewrites if needed. (TODO: fix for BigTIFF + big-endian.)
	for _, s := range strips {
		b := make([]byte, 4)
		binary.LittleEndian.PutUint32(b, uint32(len(s)))
		stripByteCounts = append(stripByteCounts, b...)
	}

	// StripOffsets: one entry per strip, placeholders (will be filled by emitter).
	stripOffsetsPlaceholder := make([]byte, len(strips)*4)

	// Inline value encoder helper: returns a 4/8 byte buffer with value v
	// encoded at width appropriate for Type. We build small SHORT/LONG
	// inlines here in little-endian; the emitter converts to file order.
	shortLE := func(v uint16) []byte {
		b := make([]byte, 4)
		binary.LittleEndian.PutUint16(b, v)
		return b
	}
	longLE := func(v uint32) []byte {
		b := make([]byte, 4)
		binary.LittleEndian.PutUint32(b, v)
		return b
	}

	// BitsPerSample: 3 × SHORT = 6 bytes — out-of-line in classic TIFF
	// (value field only 4 bytes). Emitter stores it out-of-line.
	bps := make([]byte, 6)
	binary.LittleEndian.PutUint16(bps[0:2], 8)
	binary.LittleEndian.PutUint16(bps[2:4], 8)
	binary.LittleEndian.PutUint16(bps[4:6], 8)

		{Tag: tifflow.TagPhotometric, Type: tifflow.TypeShort, Count: 1, Inline: true, Bytes: shortLE(2)},

		{Tag: tifflow.TagSamplesPerPixel, Type: tifflow.TypeShort, Count: 1, Inline: true, Bytes: shortLE(3)},
```

This is the biggest task in the plan. The splice primitive:

1. Computes the cutoff from `File.Ranges`, failing if any earlier IFD has ranges past cutoff.
2. Copies `[0, cutoff)` from input to output.
3. Emits the replacement IFD: out-of-line blobs first, strip data, then the IFD record with filled-in offsets.
4. Re-emits each tail IFD (after the label) with its internal offsets rebased.
5. Patches the predecessor's `next-IFD` pointer in the already-copied prefix.

**Splice modes:**
- `SpliceReplace`: replace IFD at index `targetIdx` with `rep`.
- `SpliceInsertBefore`: insert `rep` before IFD at index `targetIdx` (used for add-label-before-macro).
- `SpliceAppend`: append `rep` after the last IFD (used for add-label at end of chain).
- `SpliceRemove`: remove IFD at index `targetIdx` (used for strip-label).

```go
// doSplice implements Splice.
func doSplice(p SpliceParams) error {
	// 1. Determine which IFDs belong to the tail (re-emitted) vs the prefix (copied).
	//    Compute the "cutoff" offset: min start-offset of any byte owned by any
	//    tail IFD.
	//
	//    Mode semantics:
	//      Replace:       prefix = [0, idx), tail = [idx+1..), new label at cutoff.
	//      InsertBefore:  prefix = [0, idx), tail = [idx..),   new label at cutoff.
	//      Append:        prefix = [0, n),   tail = [],        new label at EOF.
	//      Remove:        prefix = [0, idx), tail = [idx+1..), no new label.

	n := len(p.File.IFDs)
	var firstTail int
	switch p.Mode {
	case SpliceReplace:
		if p.TargetIdx < 0 || p.TargetIdx >= n {
			return fmt.Errorf("%w: target idx out of range", ErrUnexpectedLayout)
		}
		firstTail = p.TargetIdx + 1
	case SpliceInsertBefore:
		if p.TargetIdx < 0 || p.TargetIdx > n {
			return fmt.Errorf("%w: target idx out of range", ErrUnexpectedLayout)
		}
		firstTail = p.TargetIdx
	case SpliceAppend:
		firstTail = n
	case SpliceRemove:
		if p.TargetIdx < 0 || p.TargetIdx >= n {
			return fmt.Errorf("%w: target idx out of range", ErrUnexpectedLayout)
		}
		firstTail = p.TargetIdx + 1
	default:
		return fmt.Errorf("unknown splice mode %d", p.Mode)
	}

	// Also include the IFD being replaced/removed itself in the tail
	// for range-dominance checking (its bytes also shouldn't appear in prefix).
	minTailOwner := firstTail
	if p.Mode == SpliceReplace || p.Mode == SpliceRemove {
		minTailOwner = p.TargetIdx
	}

	var cutoff uint64
	inInfo, err := os.Stat(p.InPath)
	if err != nil {
		return fmt.Errorf("stat input: %w", err)
	}
	fileSize := uint64(inInfo.Size())

	if minTailOwner >= n {
		// No tail, no IFD to replace.
		cutoff = fileSize
	} else {
		cutoff = p.File.Ranges.MinOffsetOfOwnersAtOrAfter(minTailOwner)
	}

	// 2. Validate: no earlier IFD has any range at or after cutoff.
	for i := 0; i < minTailOwner; i++ {
		if r, ok := p.File.Ranges.AnyRangeOfOwnerAtOrAfter(i, cutoff); ok {
			return fmt.Errorf("%w: IFD %d has range %q at offset %d, past cutoff %d",
				ErrUnexpectedLayout, i, r.What, r.Start, cutoff)
		}
	}

	// 3. Open input read-only and output temp.
	in, err := os.Open(p.InPath)
	if err != nil {
		return fmt.Errorf("open input: %w", err)
	}
	defer in.Close()

	outTmp := p.OutPath + fmt.Sprintf(".tmp-%d", os.Getpid())
	out, err := os.OpenFile(outTmp, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0o644)
	if err != nil {
		return fmt.Errorf("create output: %w", err)
	}
	committed := false
	defer func() {
		out.Close()
		if !committed {
			os.Remove(outTmp)
		}
	}()

	// 4. Copy prefix [0, cutoff).
	if _, err := io.Copy(out, io.NewSectionReader(in, 0, int64(cutoff))); err != nil {
		return fmt.Errorf("copy prefix: %w", err)
	}

	// 5. Emit new IFD (if any) + tail IFDs.
	//    Keep track of each emitted IFD's record offset and its "next-IFD
	//    pointer location" so we can chain them.
	var emittedIFDOffsets []uint64
	var emittedIFDNextPtrLocs []uint64

	// Helper: emit a ReplacementIFD at the current output offset. Returns the
	// IFD's record start offset.
	emitReplacement := func(rep *ReplacementIFD) (uint64, uint64, error) {
		return emitReplacementIFD(out, p.File.Header, rep)
	}

	// Helper: re-emit an existing IFD, rebasing any out-of-line data.
	reemitExisting := func(idx int) (uint64, uint64, error) {
		return reemitIFD(out, in, p.File, idx)
	}

	switch p.Mode {
	case SpliceReplace, SpliceInsertBefore:
		recOff, nextPtrLoc, err := emitReplacement(p.Replacement)
		if err != nil {
			return err
		}
		emittedIFDOffsets = append(emittedIFDOffsets, recOff)
		emittedIFDNextPtrLocs = append(emittedIFDNextPtrLocs, nextPtrLoc)
	case SpliceAppend:
		// Emit the replacement as the last IFD.
		recOff, nextPtrLoc, err := emitReplacement(p.Replacement)
		if err != nil {
			return err
		}
		emittedIFDOffsets = append(emittedIFDOffsets, recOff)
		emittedIFDNextPtrLocs = append(emittedIFDNextPtrLocs, nextPtrLoc)
	case SpliceRemove:
		// No new IFD.
	}

	for i := firstTail; i < n; i++ {
		recOff, nextPtrLoc, err := reemitExisting(i)
		if err != nil {
			return err
		}
		emittedIFDOffsets = append(emittedIFDOffsets, recOff)
		emittedIFDNextPtrLocs = append(emittedIFDNextPtrLocs, nextPtrLoc)
	}

	// 6. Patch the chain:
	//    - Emitted IFDs' next-IFD pointers link in order, last → 0.
	//    - The predecessor's next-IFD pointer in the prefix points to the
	//      first emitted IFD (or 0 if there are no emitted IFDs).
	//
	// Compute predecessor: the IFD in the prefix whose next-IFD originally
	// led into the tail-or-replaced. For Append mode, predecessor is IFD n-1.
	valueFieldSize := 4
	if p.File.Header.BigTIFF {
		valueFieldSize = 8
	}
	writeOffsetAt := func(f *os.File, loc uint64, val uint64) error {
		buf := make([]byte, valueFieldSize)
		if p.File.Header.BigTIFF {
			p.File.Header.ByteOrder.PutUint64(buf, val)
		} else {
			p.File.Header.ByteOrder.PutUint32(buf, uint32(val))
		}
		_, err := f.WriteAt(buf, int64(loc))
		return err
	}

	// Link emitted IFDs next-to-next.
	for i := 0; i < len(emittedIFDOffsets); i++ {
		var nextVal uint64
		if i+1 < len(emittedIFDOffsets) {
			nextVal = emittedIFDOffsets[i+1]
		}
		if err := writeOffsetAt(out, emittedIFDNextPtrLocs[i], nextVal); err != nil {
			return fmt.Errorf("patch emitted next-IFD: %w", err)
		}
	}

	// Patch predecessor.
	var predecessorNextPtrLoc uint64
	var predecessorNextVal uint64
	switch p.Mode {
	case SpliceReplace, SpliceInsertBefore, SpliceRemove:
		if p.TargetIdx == 0 {
			// Predecessor is the header.
			predecessorNextPtrLoc = headerFirstIFDOffsetLoc(p.File.Header)
		} else {
			predecessorNextPtrLoc = p.File.IFDs[p.TargetIdx-1].NextPointerOffset
		}
	case SpliceAppend:
		if n == 0 {
			predecessorNextPtrLoc = headerFirstIFDOffsetLoc(p.File.Header)
		} else {
			predecessorNextPtrLoc = p.File.IFDs[n-1].NextPointerOffset
		}
	}
	if len(emittedIFDOffsets) > 0 {
		predecessorNextVal = emittedIFDOffsets[0]
	}
	if err := writeOffsetAt(out, predecessorNextPtrLoc, predecessorNextVal); err != nil {
		return fmt.Errorf("patch predecessor next-IFD: %w", err)
	}

	// 7. fsync + rename.
	if p.Fsync {
		if err := out.Sync(); err != nil {
			return fmt.Errorf("fsync: %w", err)
		}
	}
	if err := out.Close(); err != nil {
		return fmt.Errorf("close output: %w", err)
	}
	if err := os.Rename(outTmp, p.OutPath); err != nil {
		return fmt.Errorf("rename: %w", err)
	}
	committed = true
	return nil
}

func headerFirstIFDOffsetLoc(h *Header) uint64 {
	// In classic TIFF, first-IFD offset is at byte 4 (4 bytes).
	// In BigTIFF, first-IFD offset is at byte 8 (8 bytes).
	if h.BigTIFF {
		return 8
	}
	return 4
}

// emitReplacementIFD writes a ReplacementIFD to out at the current seek
// position. Returns the IFD record's offset and the next-IFD pointer offset.
//
// Layout emitted (at current position):
//   - All out-of-line tag blobs (in tag order, except StripOffsets which comes
//     later — its value is written inside the IFD record).
//   - Strip data blobs.
//   - IFD record (entry count + entries + next-IFD pointer placeholder = 0).
//
// After this returns, the caller patches the next-IFD pointer.
func emitReplacementIFD(out *os.File, h *Header, rep *ReplacementIFD) (uint64, uint64, error) {
	curOff, err := out.Seek(0, io.SeekCurrent)
	if err != nil {
		return 0, 0, err
	}
	cur := uint64(curOff)

	// First pass: emit out-of-line blobs (non-StripOffsets).
	// Track the offsets written so they can be referenced from the IFD.
	outlineOff := make(map[int]uint64) // tag index → offset
	for i, t := range rep.Tags {
		if t.Inline {
			continue
		}
		if t.Tag == TagStripOffsets {
			// The StripOffsets array is filled later with per-strip offsets.
			continue
		}
		outlineOff[i] = cur
		if _, err := out.Write(t.Bytes); err != nil {
			return 0, 0, err
		}
		cur += uint64(len(t.Bytes))
		// Word-align.
		if cur%2 != 0 {
			if _, err := out.Write([]byte{0}); err != nil {
				return 0, 0, err
			}
			cur++
		}
	}

	// Strip data.
	stripOffsets := make([]uint64, len(rep.StripData))
	for i, s := range rep.StripData {
		stripOffsets[i] = cur
		if _, err := out.Write(s); err != nil {
			return 0, 0, err
		}
		cur += uint64(len(s))
		if cur%2 != 0 {
			if _, err := out.Write([]byte{0}); err != nil {
				return 0, 0, err
			}
			cur++
		}
	}

	// StripOffsets value blob.
	for i, t := range rep.Tags {
		if t.Tag != TagStripOffsets {
			continue
		}
		// Build the offsets array in the file's byte order.
		buf := make([]byte, 4*len(stripOffsets))
		for j, so := range stripOffsets {
			h.ByteOrder.PutUint32(buf[j*4:], uint32(so))
		}
		// Re-assign Bytes and mark as not-yet-emitted (we emit now).
		rep.Tags[i].Bytes = buf
		outlineOff[i] = cur
		if _, err := out.Write(buf); err != nil {
			return 0, 0, err
		}
		cur += uint64(len(buf))
		if cur%2 != 0 {
			if _, err := out.Write([]byte{0}); err != nil {
				return 0, 0, err
			}
			cur++
		}
	}

	// IFD record starts here.
	ifdOff := cur
	entryCount := uint64(len(rep.Tags))
	countSize := 2
	if h.BigTIFF {
		countSize = 8
	}
	entrySize := 12
	if h.BigTIFF {
		entrySize = 20
	}
	valueFieldSize := 4
	if h.BigTIFF {
		valueFieldSize = 8
	}

	// Entry count.
	if h.BigTIFF {
		if err := binary.Write(out, h.ByteOrder, entryCount); err != nil {
			return 0, 0, err
		}
	} else {
		if err := binary.Write(out, h.ByteOrder, uint16(entryCount)); err != nil {
			return 0, 0, err
		}
	}

	// Entries.
	for i, t := range rep.Tags {
		if err := binary.Write(out, h.ByteOrder, t.Tag); err != nil {
			return 0, 0, err
		}
		if err := binary.Write(out, h.ByteOrder, uint16(t.Type)); err != nil {
			return 0, 0, err
		}
		if h.BigTIFF {
			if err := binary.Write(out, h.ByteOrder, t.Count); err != nil {
				return 0, 0, err
			}
		} else {
			if err := binary.Write(out, h.ByteOrder, uint32(t.Count)); err != nil {
				return 0, 0, err
			}
		}
		valBuf := make([]byte, valueFieldSize)
		if t.Inline {
			// Convert LE-encoded Inline Bytes to file byte order.
			copy(valBuf, convertInlineEndian(t.Type, t.Count, t.Bytes, h.ByteOrder, valueFieldSize))
		} else {
			off := outlineOff[i]
			if h.BigTIFF {
				h.ByteOrder.PutUint64(valBuf, off)
			} else {
				h.ByteOrder.PutUint32(valBuf, uint32(off))
			}
		}
		if _, err := out.Write(valBuf); err != nil {
			return 0, 0, err
		}
	}

	// next-IFD placeholder = 0.
	nextLoc := ifdOff + uint64(countSize) + entryCount*uint64(entrySize)
	zeroNext := make([]byte, valueFieldSize)
	if _, err := out.Write(zeroNext); err != nil {
		return 0, 0, err
	}

	return ifdOff, nextLoc, nil
}

// convertInlineEndian takes an inline value field encoded little-endian
// (as BuildLabelIFD produces) and re-encodes it in targetOrder. Only the
// first Count * Type.Size() bytes are significant.
func convertInlineEndian(tp TagType, count uint64, leBytes []byte, order binary.ByteOrder, valueFieldSize int) []byte {
	out := make([]byte, valueFieldSize)
	sz := tp.Size()
	total := int(count) * sz
	if total > valueFieldSize {
		total = valueFieldSize
	}
	for i := 0; i < int(count); i++ {
		b := leBytes[i*sz : (i+1)*sz]
		var v uint64
		switch sz {
		case 1:
			v = uint64(b[0])
		case 2:
			v = uint64(binary.LittleEndian.Uint16(b))
		case 4:
			v = uint64(binary.LittleEndian.Uint32(b))
		case 8:
			v = binary.LittleEndian.Uint64(b)
		}
		target := out[i*sz : (i+1)*sz]
		switch sz {
		case 1:
			target[0] = byte(v)
		case 2:
			order.PutUint16(target, uint16(v))
		case 4:
			order.PutUint32(target, uint32(v))
		case 8:
			order.PutUint64(target, v)
		}
	}
	return out
}

// reemitIFD re-emits IFD idx from file into out at the current seek position,
// rebasing its out-of-line data and strip data. Returns the new IFD record
// offset and next-IFD pointer offset.
func reemitIFD(out *os.File, in *os.File, file *File, idx int) (uint64, uint64, error) {
	ifd := file.IFDs[idx]
	h := file.Header

	// Emit out-of-line data at the current position, tracking old→new offsets.
	curPos, err := out.Seek(0, io.SeekCurrent)
	if err != nil {
		return 0, 0, err
	}
	cur := uint64(curPos)

	newOutlineOff := make(map[int]uint64) // entry index → new offset
	for i, e := range ifd.Entries {
		if e.Inline || e.DataSize == 0 || e.Tag == TagStripOffsets || e.Tag == TagTileOffsets {
			continue
		}
		blob := make([]byte, e.DataSize)
		if _, err := in.ReadAt(blob, int64(e.DataOffset)); err != nil {
			return 0, 0, fmt.Errorf("read out-of-line tag %d: %w", e.Tag, err)
		}
		newOutlineOff[i] = cur
		if _, err := out.Write(blob); err != nil {
			return 0, 0, err
		}
		cur += uint64(len(blob))
		if cur%2 != 0 {
			_, _ = out.Write([]byte{0})
			cur++
		}
	}

	// Copy strip/tile data and build new offsets array.
	var stripTag, countTag uint16
	oldOffsets, ok := ifd.UintArray(TagStripOffsets, h.ByteOrder)
	if ok {
		stripTag, countTag = TagStripOffsets, TagStripByteCounts
	} else {
		oldOffsets, ok = ifd.UintArray(TagTileOffsets, h.ByteOrder)
		stripTag, countTag = TagTileOffsets, TagTileByteCounts
	}
	var newOffsetsBuf []byte
	if ok {
		counts, _ := ifd.UintArray(countTag, h.ByteOrder)
		newOffsets := make([]uint64, len(oldOffsets))
		for i, off := range oldOffsets {
			cnt := counts[i]
			if cnt == 0 {
				newOffsets[i] = 0
				continue
			}
			newOffsets[i] = cur
			buf := make([]byte, cnt)
			if _, err := in.ReadAt(buf, int64(off)); err != nil {
				return 0, 0, fmt.Errorf("read strip: %w", err)
			}
			if _, err := out.Write(buf); err != nil {
				return 0, 0, err
			}
			cur += cnt
			if cur%2 != 0 {
				_, _ = out.Write([]byte{0})
				cur++
			}
		}
		// Use the same width as the original offsets tag type.
		var entryIdx int = -1
		for i, e := range ifd.Entries {
			if e.Tag == stripTag {
				entryIdx = i
				break
			}
		}
		_ = entryIdx
		entry := findEntry(ifd, stripTag)
		if entry != nil && (entry.Type == TypeLong8 || entry.Type == TypeIFD8) {
			newOffsetsBuf = make([]byte, 8*len(newOffsets))
			for i, v := range newOffsets {
				h.ByteOrder.PutUint64(newOffsetsBuf[i*8:], v)
			}
		} else {
			newOffsetsBuf = make([]byte, 4*len(newOffsets))
			for i, v := range newOffsets {
				h.ByteOrder.PutUint32(newOffsetsBuf[i*4:], uint32(v))
			}
		}
	}
	// Write the new strip/tile offsets blob out-of-line (if it doesn't fit inline).
	valueFieldSize := 4
	if h.BigTIFF {
		valueFieldSize = 8
	}
	if newOffsetsBuf != nil {
		for i, e := range ifd.Entries {
			if e.Tag == stripTag {
				if len(newOffsetsBuf) > valueFieldSize {
					newOutlineOff[i] = cur
					if _, err := out.Write(newOffsetsBuf); err != nil {
						return 0, 0, err
					}
					cur += uint64(len(newOffsetsBuf))
					if cur%2 != 0 {
						_, _ = out.Write([]byte{0})
						cur++
					}
				}
				break
			}
		}
	}

	// Emit the IFD record.
	ifdOff := cur
	countSize := 2
	if h.BigTIFF {
		countSize = 8
	}
	entrySize := 12
	if h.BigTIFF {
		entrySize = 20
	}

	if h.BigTIFF {
		_ = binary.Write(out, h.ByteOrder, uint64(len(ifd.Entries)))
	} else {
		_ = binary.Write(out, h.ByteOrder, uint16(len(ifd.Entries)))
	}
	for i, e := range ifd.Entries {
		_ = binary.Write(out, h.ByteOrder, e.Tag)
		_ = binary.Write(out, h.ByteOrder, uint16(e.Type))
		if h.BigTIFF {
			_ = binary.Write(out, h.ByteOrder, e.Count)
		} else {
			_ = binary.Write(out, h.ByteOrder, uint32(e.Count))
		}
		valBuf := make([]byte, valueFieldSize)
		switch {
		case e.Tag == stripTag && newOffsetsBuf != nil && len(newOffsetsBuf) <= valueFieldSize:
			copy(valBuf, newOffsetsBuf)
		case e.Inline:
			copy(valBuf, e.RawValueField)
		default:
			// Out-of-line: new offset.
			if newOff, ok := newOutlineOff[i]; ok {
				if h.BigTIFF {
					h.ByteOrder.PutUint64(valBuf, newOff)
				} else {
					h.ByteOrder.PutUint32(valBuf, uint32(newOff))
				}
			} else {
				// Shouldn't happen — diagnostic.
				return 0, 0, fmt.Errorf("%w: no new offset for tag %d on IFD %d", ErrUnexpectedLayout, e.Tag, idx)
			}
		}
		if _, err := out.Write(valBuf); err != nil {
			return 0, 0, err
		}
	}

	cur = ifdOff + uint64(countSize) + uint64(len(ifd.Entries))*uint64(entrySize)
	nextLoc := cur
	zeroNext := make([]byte, valueFieldSize)
	if _, err := out.Write(zeroNext); err != nil {
		return 0, 0, err
	}
	return ifdOff, nextLoc, nil
}

func findEntry(ifd *IFD, tag uint16) *IFDEntry {
	for i := range ifd.Entries {
		if ifd.Entries[i].Tag == tag {
			return &ifd.Entries[i]
		}
	}
	return nil
}
```

```go
package aperio

import (
	"fmt"
	"image"
	"os"

	"github.com/tcornish/wsi-label/internal/tifflow"
	"github.com/tcornish/wsi-label/internal/wsi"
)

type backend struct{}

// New returns an Aperio wsi.Backend.
func New() wsi.Backend { return &backend{} }

func (backend) Name() string { return "aperio" }

func (backend) Detect(path string) (bool, error) { return Detect(path) }

func (backend) ReplaceLabel(inPath, outPath string, img image.Image, strictReplace, fsync bool) (*wsi.ReplaceResult, error) {
	f, err := os.Open(inPath)
	if err != nil {
		return nil, fmt.Errorf("open: %w", err)
	}
	defer f.Close()
	file, err := tifflow.Parse(f)
	if err != nil {
		return nil, err
	}

	labelIdx := FindLabelIFD(file)
	resized := FitTo(img, AperioLabelWidth, AperioLabelHeight, AperioLabelBG)
	rep := BuildLabelIFD(resized)

	params := tifflow.SpliceParams{
		InPath: inPath, OutPath: outPath, File: file,
		Replacement: rep, Fsync: fsync,
	}
	result := &wsi.ReplaceResult{}

	if labelIdx >= 0 {
		params.Mode = tifflow.SpliceReplace
		params.TargetIdx = labelIdx
	} else {
		if strictReplace {
			return nil, fmt.Errorf("aperio: %w", wsi.ErrNoLabel)
		}
		result.Added = true
		macroIdx := FindMacroIFD(file)
		if macroIdx >= 0 {
			params.Mode = tifflow.SpliceInsertBefore
			params.TargetIdx = macroIdx
		} else {
			params.Mode = tifflow.SpliceAppend
		}
	}

	if err := tifflow.Splice(params); err != nil {
		return nil, err
	}
	return result, nil
}

func (backend) StripLabel(inPath, outPath string, fsync bool) error {
	f, err := os.Open(inPath)
	if err != nil {
		return fmt.Errorf("open: %w", err)
```

```go
	if err := fs.Parse(args); err != nil {

	}

		return fmt.Errorf("%w: replace takes <wsi-in> <label-image> [<wsi-out>]", ErrUsage)
	}
	in := fs.Arg(0)
	labelPath := fs.Arg(1)
	outExplicit := ""
	if fs.NArg() == 3 {
		outExplicit = fs.Arg(2)
	}

	out, err := ResolveOutputPath(in, outExplicit, *overwrite)
	if err != nil {
		return err
	}

	// Decode label.
	lf, err := os.Open(labelPath)
	if err != nil {
		return fmt.Errorf("%w: label: %v", ErrUnreadable, err)
	}
	defer lf.Close()
	img, _, err := image.Decode(lf)
	if err != nil {
		return fmt.Errorf("%w: decode label: %v", ErrUnreadable, err)

	bgColor, err := parseHexRGB(*bg)
	if err != nil {
		return fmt.Errorf("%w: --bg: %v", ErrUsage, err)
	}
	_ = bgColor // currently aperio.AperioLabelBG is used; future: thread through.

	// Dispatch to aperio backend (registry has one entry today).
	b := aperio.New()
	ok, derr := b.Detect(in)
	if derr != nil {
		return fmt.Errorf("%w: %v", ErrUnreadable, derr)
	}
	if !ok {
		return fmt.Errorf("%w: not an Aperio SVS", ErrFormatUnrecognized)
	}
	res, err := b.ReplaceLabel(in, out, img, *strict, *fsyncFlag)
	if err != nil {
		return err
	}
	if res.Added {
		if !*quiet {
			fmt.Fprintf(stderr, "wsi-label: no existing label; added one to %s\n", out)
		}
		return fmt.Errorf("%w: %s", ErrLabelAdded, out)
	}
	if !*quiet {
		fmt.Fprintf(stdout, "wsi-label: replaced label: %s → %s\n", in, out)
	}
	_ = wsi.ErrNoLabel // keep import live for error-type checks in Run
	return nil
}

func parseHexRGB(s string) (struct{ R, G, B uint8 }, error) {
	var rgb struct{ R, G, B uint8 }
	s = strings.TrimPrefix(s, "#")
	if len(s) != 6 {
		return rgb, fmt.Errorf("expected 6 hex digits, got %q", s)
	}
	n, err := strconv.ParseUint(s, 16, 32)
	if err != nil {
		return rgb, err
	}
	rgb.R = uint8(n >> 16)
	rgb.G = uint8(n >> 8)
	rgb.B = uint8(n)
	return rgb, nil
}
```

```python
#!/usr/bin/env python3
"""Cross-validate wsi-label output against openslide + tifffile.

For each sample SVS in sample_svs/, runs `wsi-label replace` and then:
  - Opens with openslide; asserts associated_images["label"] is present and
    matches our input label (modulo resize).
  - Opens with tifffile; asserts pyramid level dimensions identical to input.

Usage:
    python scripts/cross_validate.py path/to/wsi-label
"""

import hashlib
import subprocess
import sys
from pathlib import Path

import numpy as np
import openslide
import tifffile
from PIL import Image

HERE = Path(__file__).resolve().parent.parent
SAMPLES = HERE / "sample_svs"


def pyramid_dims(path: Path) -> list[tuple[int, int]]:
    with tifffile.TiffFile(str(path)) as tf:
        dims = []
        for page in tf.pages:
            if page.is_tiled:
                dims.append((page.imagewidth, page.imagelength))
    return dims


def make_label(tmpdir: Path) -> Path:
    img = Image.new("RGB", (600, 400), (200, 150, 100))
    p = tmpdir / "label.png"
    img.save(p)
    return p


def validate_one(exe: Path, sample: Path, label: Path, tmpdir: Path) -> tuple[bool, str]:
    out = tmpdir / f"out_{sample.name}"
    result = subprocess.run(
        [str(exe), "replace", "--overwrite", str(sample), str(label), str(out)],
        capture_output=True,
    )
    if result.returncode not in (0, 10):
        return False, f"exit {result.returncode}: {result.stderr.decode()}"

    in_dims = pyramid_dims(sample)
    out_dims = pyramid_dims(out)
    if in_dims != out_dims:
        return False, f"pyramid dims differ: {in_dims} vs {out_dims}"

    try:
        slide = openslide.OpenSlide(str(out))
    except openslide.OpenSlideError as e:
        return False, f"openslide open: {e}"
    if "label" not in slide.associated_images:
        return False, "output has no label associated image"
    got = np.asarray(slide.associated_images["label"].convert("RGB"))
    if got.shape[0] == 0 or got.shape[1] == 0:
        return False, "empty label"
    return True, "ok"


def main() -> int:
    if len(sys.argv) < 2:
        print("usage: cross_validate.py <wsi-label-binary>", file=sys.stderr)
        return 2
    exe = Path(sys.argv[1]).resolve()
    tmpdir = HERE / ".tmp_crossvalidate"
    tmpdir.mkdir(exist_ok=True)
    label = make_label(tmpdir)

    failures = 0
    samples = sorted(SAMPLES.glob("*.svs"))
    for s in samples:
        ok, msg = validate_one(exe, s, label, tmpdir)
        status = "OK " if ok else "FAIL"
        print(f"[{status}] {s.name}: {msg}")
        if not ok:
            failures += 1

    print(f"\n{len(samples) - failures}/{len(samples)} samples passed.")
    return 1 if failures else 0


if __name__ == "__main__":
    sys.exit(main())
```
